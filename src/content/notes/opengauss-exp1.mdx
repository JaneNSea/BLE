### 配置字符集

openGauss 要求系统字符集为 UTF-8。登录服务器后，先查看当前字符集配置：

```bash
echo $LANG
locale
```

若未配置为 UTF-8，需修改 `/etc/locale.conf`：

```bash
cat >>/etc/profile<<EOF
export LANG=en_US.UTF‐8
EOF
```

使配置生效：

```bash
source /etc/profile
```

同时确保系统已安装 UTF-8 语言包：

```bash
yum install -y glibc-locale-source
localedef -c -f UTF-8 -i en_US en_US.UTF-8
```

### 创建 XML 配置文件

openGauss 使用 XML 文件定义集群拓扑。创建 `clusterconfig.xml`：

```bash
vi /opt/software/openGauss/clusterconfig.xml
```

单机节点配置示例：

```xml
<?xml version="1.0" encoding="UTF-8"?> 
<ROOT> 
    <CLUSTER> 
        <PARAM name="clusterName" value="dbCluster" /> 
        <PARAM name="nodeNames" value="ecs-d3cb" /> 
        <PARAM name="backIp1s" value="192.168.0.189"/> 
        <PARAM name="gaussdbAppPath" value="/opt/gaussdb/app" /> 
        <PARAM name="gaussdbLogPath" value="/var/log/gaussdb" /> 
        <PARAM name="gaussdbToolPath" value="/opt/huawei/wisequery" /> 
        <PARAM name="corePath" value="/opt/opengauss/corefile"/> 
        <PARAM name="clusterType" value="single-inst"/> 
    </CLUSTER> 
   
    <DEVICELIST> 
        
        <DEVICE sn="1000001"> 
            <PARAM name="name" value="ecs-d3cb"/> 
            <PARAM name="azName" value="AZ1"/> 
            <PARAM name="azPriority" value="1"/> 
            <PARAM name="backIp1" value="192.168.0.189"/> 
            <PARAM name="sshIp1" value="192.168.0.189"/> 
             
	    <!--dbnode--> 
	    <PARAM name="dataNum" value="1"/> 
	    <PARAM name="dataPortBase" value="26000"/> 
	    <PARAM name="dataNode1" value="/gaussdb/data/db1"/> 
        </DEVICE> 
    </DEVICELIST> 
</ROOT>
```

其中 `backIp1` 填写服务器内网 IP。

```bash
# 允许所有地址连接
listen_addresses = '*'
```

```bash

# TYPE  DATABASE    USER    ADDRESS       METHOD
host    all         all     0.0.0.0/0     md5
```

### 创建数据库表

在 `school_management` 数据库中创建学籍与成绩管理系统的 6 张表：

```sql
-- ============================================
-- 1. 学院表 (xyb) - 先创建，因为被其他表引用
-- ============================================
CREATE TABLE xyb (
    ydh  CHAR(2)  NOT NULL,  -- 学院代号
    ymc  VARCHAR(30) NOT NULL,  -- 学院名称
    CONSTRAINT pk_xyb PRIMARY KEY (ydh)
);

COMMENT ON TABLE xyb IS '学院表';
COMMENT ON COLUMN xyb.ydh IS '学院代号';
COMMENT ON COLUMN xyb.ymc IS '学院名称';

-- ============================================
-- 2. 学生表 (xs)
-- ============================================
CREATE TABLE xs (
    xm   VARCHAR(8)  NOT NULL,  -- 姓名
    xh   CHAR(10)    NOT NULL,  -- 学号
    ydh  CHAR(2),                 -- 所属学院代号 (FK -> xyb)
    bj   VARCHAR(8),              -- 班级
    chrq DATE,                    -- 出生日期
    xb   CHAR(2),                 -- 性别
    CONSTRAINT pk_xs PRIMARY KEY (xh),
    CONSTRAINT fk_xs_xyb FOREIGN KEY (ydh) REFERENCES xyb(ydh)
);

COMMENT ON TABLE xs IS '学生表';
COMMENT ON COLUMN xs.xm IS '姓名';
COMMENT ON COLUMN xs.xh IS '学号';
COMMENT ON COLUMN xs.ydh IS '所属学院代号';
COMMENT ON COLUMN xs.bj IS '班级';
COMMENT ON COLUMN xs.chrq IS '出生日期';
COMMENT ON COLUMN xs.xb IS '性别';

-- ============================================
-- 3. 课程表 (kc)
-- ============================================
CREATE TABLE kc (
    kcbh CHAR(3)     NOT NULL,  -- 课程编号
    kc   VARCHAR(20) NOT NULL,  -- 课程名称
    lx   VARCHAR(10),             -- 课程类型（必修、选修、任选）
    xf   NUMERIC(4,1),            -- 学分（总长5位含小数点，小数1位 → PostgreSQL用NUMERIC(4,1)，即整数3位+小数1位）
    CONSTRAINT pk_kc PRIMARY KEY (kcbh)
);

COMMENT ON TABLE kc IS '课程表';
COMMENT ON COLUMN kc.kcbh IS '课程编号';
COMMENT ON COLUMN kc.kc IS '课程名称';
COMMENT ON COLUMN kc.lx IS '课程类型';
COMMENT ON COLUMN kc.xf IS '学分';

-- ============================================
-- 4. 教师表 (js)
-- ============================================
CREATE TABLE js (
    xm   VARCHAR(8)  NOT NULL,  -- 姓名
    jsbh CHAR(10)    NOT NULL,  -- 教师编号
    zc   VARCHAR(6),              -- 职称
    ydh  CHAR(2),                 -- 所属学院代号 (FK -> xyb)
    CONSTRAINT pk_js PRIMARY KEY (jsbh),
    CONSTRAINT fk_js_xyb FOREIGN KEY (ydh) REFERENCES xyb(ydh)
);

COMMENT ON TABLE js IS '教师表';
COMMENT ON COLUMN js.xm IS '姓名';
COMMENT ON COLUMN js.jsbh IS '教师编号';
COMMENT ON COLUMN js.zc IS '职称';
COMMENT ON COLUMN js.ydh IS '所属学院代号';

-- ============================================
-- 5. 授课表 (sk) - 关联课程与教师
-- ============================================
CREATE TABLE sk (
    kcbh CHAR(3)  NOT NULL,  -- 课程编号 (FK -> kc)
    bh   CHAR(10) NOT NULL,  -- 教师编号 (FK -> js)
    CONSTRAINT pk_sk PRIMARY KEY (kcbh, bh),
    CONSTRAINT fk_sk_kc FOREIGN KEY (kcbh) REFERENCES kc(kcbh),
    CONSTRAINT fk_sk_js FOREIGN KEY (bh) REFERENCES js(jsbh)
);

COMMENT ON TABLE sk IS '授课表';
COMMENT ON COLUMN sk.kcbh IS '课程编号';
COMMENT ON COLUMN sk.bh IS '教师编号';

-- ============================================
-- 6. 学生选课表 (xk) - 关联学生、课程、教师、成绩
-- ============================================
CREATE TABLE xk (
    xh   CHAR(10) NOT NULL,  -- 学号 (FK -> xs)
    kcbh CHAR(3)  NOT NULL,  -- 课程编号 (FK -> kc)
    jsbh CHAR(10) NOT NULL,  -- 教师编号 (FK -> js)
    cj   NUMERIC(4,1),        -- 成绩
    CONSTRAINT pk_xk PRIMARY KEY (xh, kcbh, jsbh),
    CONSTRAINT fk_xk_xs FOREIGN KEY (xh) REFERENCES xs(xh),
    CONSTRAINT fk_xk_kc FOREIGN KEY (kcbh) REFERENCES kc(kcbh),
    CONSTRAINT fk_xk_js FOREIGN KEY (jsbh) REFERENCES js(jsbh)
);

COMMENT ON TABLE xk IS '学生选课表';
COMMENT ON COLUMN xk.xh IS '学号';
COMMENT ON COLUMN xk.kcbh IS '课程编号';
COMMENT ON COLUMN xk.jsbh IS '教师编号';
COMMENT ON COLUMN xk.cj IS '成绩';

-- ============================================
-- 索引创建（提高查询效率）
-- ============================================

-- 学生表：按学院查询频繁，为外键 ydh 建立索引
CREATE INDEX idx_xs_ydh ON xs(ydh);

-- 教师表：按学院查询频繁，为外键 ydh 建立索引
CREATE INDEX idx_js_ydh ON js(ydh);

-- 授课表：两个外键分别建立索引，便于按课程或教师查询
CREATE INDEX idx_sk_kcbh ON sk(kcbh);
CREATE INDEX idx_sk_bh ON sk(bh);

-- 学生选课表：为各外键建立索引，便于关联查询和统计
CREATE INDEX idx_xk_xh ON xk(xh);
CREATE INDEX idx_xk_kcbh ON xk(kcbh);
CREATE INDEX idx_xk_jsbh ON xk(jsbh);

-- 成绩查询频繁，为成绩字段建立索引（用于成绩统计、排名等）
CREATE INDEX idx_xk_cj ON xk(cj);
```

```bash
# 删除旧的网卡配置文件（设备名变了，旧配置已经失效）
rm -f /etc/sysconfig/network-scripts/ifcfg-ens*
# 重启网络服务，让系统自动生成新配置
systemctl restart NetworkManager
# 强制让新网卡获取IP
nmcli connection add type ethernet con-name ens33 ifname ens33 autoconnect yes ipv4.method auto
nmcli connection up ens33
#这一步是改掉监听端口，不然用远程连接工具没法后续启动高斯数据库
vi /gaussdb/data/db1/postgresql.conf
#进去将原本的配置改掉
listen_addresses = '0.0.0.0'
```
